cmake_minimum_required(VERSION 3.20)
project(distributed_clustering LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(HPC_ENABLE_CUDA "Build NVIDIA CUDA implementations when available" ON)
option(HPC_ENABLE_MPI "Build distributed clustering when MPI is available" ON)

add_library(distributed_clustering_ops SHARED native/ops.cpp)
target_include_directories(distributed_clustering_ops PUBLIC native)
target_compile_options(distributed_clustering_ops PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-O3>)
if(APPLE)
  find_path(HPC_OMP_INCLUDE omp.h HINTS /opt/homebrew/opt/libomp/include /usr/local/opt/libomp/include)
  find_library(HPC_OMP_LIBRARY omp HINTS /opt/homebrew/opt/libomp/lib /usr/local/opt/libomp/lib)
  # PyTorch ships its own libomp. Reuse that exact runtime to prevent two OpenMP
  # runtimes from aborting the process when Python loads both native code and torch.
  if(HPC_PYTHON_EXECUTABLE)
    execute_process(COMMAND "${HPC_PYTHON_EXECUTABLE}" -c
      "import importlib.util,pathlib; s=importlib.util.find_spec('torch'); print(pathlib.Path(s.origin).parent/'lib/libomp.dylib' if s else '')"
      OUTPUT_VARIABLE HPC_TORCH_OMP OUTPUT_STRIP_TRAILING_WHITESPACE)
    if(EXISTS "${HPC_TORCH_OMP}")
      set(HPC_OMP_LIBRARY "${HPC_TORCH_OMP}")
      message(STATUS "Reusing PyTorch OpenMP runtime: ${HPC_OMP_LIBRARY}")
    endif()
  endif()
  if(HPC_OMP_INCLUDE AND HPC_OMP_LIBRARY)
    target_compile_options(distributed_clustering_ops PUBLIC $<$<COMPILE_LANGUAGE:CXX>:-Xpreprocessor> $<$<COMPILE_LANGUAGE:CXX>:-fopenmp>)
    target_include_directories(distributed_clustering_ops PUBLIC ${HPC_OMP_INCLUDE})
    target_link_libraries(distributed_clustering_ops PUBLIC ${HPC_OMP_LIBRARY})
    target_compile_definitions(distributed_clustering_ops PUBLIC HPC_OPENMP=1)
  endif()
else()
  find_package(OpenMP QUIET)
  if(OpenMP_CXX_FOUND)
    target_link_libraries(distributed_clustering_ops PUBLIC OpenMP::OpenMP_CXX)
    target_compile_definitions(distributed_clustering_ops PUBLIC HPC_OPENMP=1)
  endif()
endif()


if(HPC_ENABLE_CUDA)
  include(CheckLanguage)
  check_language(CUDA)
  if(CMAKE_CUDA_COMPILER)
    enable_language(CUDA)
    find_package(CUDAToolkit REQUIRED)
    add_library(distributed_clustering_cuda SHARED native/cuda_ops.cu)
    target_include_directories(distributed_clustering_cuda PUBLIC native)
    target_link_libraries(distributed_clustering_cuda PRIVATE CUDA::cudart CUDA::cublas)
    set_target_properties(distributed_clustering_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
    target_compile_options(distributed_clustering_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3>)
  else()
    message(STATUS "CUDA compiler unavailable: building CPU/OpenMP backends")
  endif()
endif()

if(HPC_ENABLE_MPI)
  find_package(MPI COMPONENTS CXX QUIET)
  if(MPI_CXX_FOUND)
    add_executable(hpc_cluster_mpi native/mpi_main.cpp)
    target_link_libraries(hpc_cluster_mpi PRIVATE distributed_clustering_ops MPI::MPI_CXX)
    if(TARGET distributed_clustering_cuda)
      target_link_libraries(hpc_cluster_mpi PRIVATE distributed_clustering_cuda)
      target_compile_definitions(hpc_cluster_mpi PRIVATE HPC_CUDA=1)
    endif()
  else()
    message(STATUS "MPI unavailable: distributed executable omitted")
  endif()
endif()

# Some PyTorch wheels retain a build-machine absolute libomp install name.
# Rewrite only our new binaries to refer to the installed wheel's runtime.
if(APPLE AND EXISTS "${HPC_TORCH_OMP}")
  execute_process(COMMAND /usr/bin/otool -D "${HPC_TORCH_OMP}" OUTPUT_VARIABLE HPC_OMP_ID_OUTPUT)
  string(REPLACE "\n" ";" HPC_OMP_ID_LINES "${HPC_OMP_ID_OUTPUT}")
  list(GET HPC_OMP_ID_LINES 1 HPC_OMP_INSTALL_NAME)
  foreach(HPC_TARGET distributed_clustering_ops hpc_cluster_mpi)
    if(TARGET ${HPC_TARGET})
      add_custom_command(TARGET ${HPC_TARGET} POST_BUILD COMMAND /usr/bin/install_name_tool
        -change "${HPC_OMP_INSTALL_NAME}" "${HPC_TORCH_OMP}" "$<TARGET_FILE:${HPC_TARGET}>")
    endif()
  endforeach()
endif()
